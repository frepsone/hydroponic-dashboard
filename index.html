  <h2>Storico Dettagliato</h2>

  <table id="tabella">

    <tr><th>Data e Ora</th><th>T. Acqua</th><th>T. Aria</th><th>Umidità</th></tr>

  </table>


  <!-- Modulo Firebase V9 -->

  <script type="module">

    import { initializeApp } from "https://www.gstatic.com/firebasejs/10.13.0/firebase-app.js";

    import { getDatabase, ref, onValue, limitToLast, query } from "https://www.gstatic.com/firebasejs/10.13.0/firebase-database.js";


    // INSERISCI QUI I TUOI DUE CODICI FIREBASE

    const firebaseConfig = {

      apiKey: "AIzaSyDTNX50TvZb-8Q7jcg8KxDyymJ8v0TY1_0",

      databaseURL: "https://hydrostradellazzo-default-rtdb.europe-west1.firebasedatabase.app"

    };


    const app = initializeApp(firebaseConfig);

    const db = getDatabase(app);

    

    // Richiediamo solo gli ultimi 100 dati per mantenere la pagina scattante

    const storicoRef = query(ref(db, 'storico'), limitToLast(100));


    // Inizializza il grafico vuoto

    let chart = new Chart(document.getElementById('grafico'), {

      type: 'line',

      data: { labels: [], datasets: [

        { label: 'Acqua (°C)', borderColor: '#36a2eb', data: [] },

        { label: 'Aria (°C)', borderColor: '#ff6384', data: [] }

      ]},

      options: { animation: false }

    });


    // Questa funzione "ascolta" Firebase. Si aggiorna DA SOLA appena l'ESP32 invia un nuovo dato!

    onValue(storicoRef, (snapshot) => {

      const data = snapshot.val();

      if (!data) return;


      const table = document.getElementById('tabella');

      // Resetta la tabella

      table.innerHTML = "<tr><th>Data e Ora</th><th>T. Acqua</th><th>T. Aria</th><th>Umidità</th></tr>";

      

      let labels = [], acqua = [], aria = [];


      // Estrai i dati dal Cloud e popola grafico e tabella

      Object.values(data).forEach(row => {

        labels.push(row.ora.substring(11, 16)); 

        acqua.push(row.t_acqua);

        aria.push(row.t_aria);

        

        let newRow = table.insertRow(1); // Inserisce in cima alla tabella

        newRow.innerHTML = `<td>${row.ora}</td><td>${row.t_acqua}°C</td><td>${row.t_aria}°C</td><td>${row.u_aria}%</td>`;

      });


      chart.data.labels = labels;

      chart.data.datasets[0].data = acqua;

      chart.data.datasets[1].data = aria;

      chart.update();

    });

  </script>
